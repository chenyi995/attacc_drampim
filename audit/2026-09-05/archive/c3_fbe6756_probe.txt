def make(corpus):
    t=CacheBlendTLB(256,'slice')
    for owner,fp,rs,kind in corpus:
        for r in rs:t.reserve(0,owner,fp,r,kind)
    t.finalize()
    return t

def locs(t,corpus):return [t.locate(0,o,fp,r,k) for o,fp,rs,k in corpus for r in rs]
def fields(a):
    x=a>>5; out={}
    for name,b in [('col',5),('row',14),('bank',2),('bg',2),('rank',1),('pch',1),('channel',4)]:
        out[name]=x&((1<<b)-1);x>>=b
    return out
