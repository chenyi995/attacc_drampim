from pathlib import Path
import hashlib,json,re,shutil,subprocess
R=Path('/data2/chenyi9/KV-PIM/attacc_drampim_822');Z=R/'audit/2026-09-05/archive'
start=json.loads((Z/'bb19f31_audit_start.json').read_text())
late=json.loads((Z/'ff5b91e_late_snapshot.json').read_text())
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
checks={}
for kind in ('protected','existing_results','paper'):
    changes=[]
    reference=late[kind] if kind=='protected' else start[kind]
    for name,value in reference.items():
        p=Path(name) if Path(name).is_absolute() else R/name
        if not p.is_file() or sha(p)!=value:changes.append(name)
    checks[kind]={'files':len(reference),'changed':changes,'unchanged':not changes,
                  'baseline':'ff5b91e after external source commits' if kind=='protected' else 'bb19f31 audit start'}
assert all(c['unchanged'] for c in checks.values()),checks
head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=R,text=True).strip();assert head==late['head']
external_source_changes=[p for p,h in start['protected'].items() if sha(R/p)!=h]
assert sorted(external_source_changes)==['src/devices.py','src/workload_runner.py','tests/test_workload.py'],external_source_changes
changed=subprocess.check_output(['git','diff','--name-only'],cwd=R,text=True).splitlines()
assert all(p.startswith(('audit/','docs/')) for p in changed),changed
untracked=subprocess.check_output(['git','ls-files','--others','--exclude-standard'],cwd=R,text=True).splitlines()
unexpected=[p for p in untracked if not p.startswith(('audit/','docs/')) and p not in start['existing_results']]
assert not unexpected,unexpected
history_changes=[p for p,h in start['prior_audit'].items() if p not in changed and sha(R/p)!=h]

bad=[];count=0
for doc in docs:
    content=doc.read_text()
    for raw in re.findall(r'\]\((<[^>]+>|[^)\n]+)\)',content):
        s=raw[1:-1] if raw.startswith('<') else raw
        if '://' in s:continue
        pathpart=s.split('#',1)[0];line=re.search(r':(\d+)$',pathpart)
        pathpart=re.sub(r':\d+$','',pathpart)
        if not pathpart:continue
        p=Path(pathpart) if pathpart.startswith('/') else doc.parent/pathpart
        count+=1
        if not p.exists():bad.append({'doc':str(doc.relative_to(R)),'target':s})
        elif line and int(line.group(1))>len(p.read_text().splitlines()):bad.append({'doc':str(doc.relative_to(R)),'line_past_end':s})
assert not bad,bad
current=(R/'audit/2026-09-05/CURRENT_ISSUES.md').read_text()
assert all(f'<a id="{x}"></a>' in current for x in ('decisions','c1','c2','c3','c4','c5','c6','c7','c8','e1','e2','contributions-check'))

assert all(abs(v['diagnostic_to_event_energy_ratio']-1)<1e-12 for v in energy['controls'])

r=controls[('A3b',0,1)];assert r['summary']['requests']['r']['first_token_s']<r['final_gpu_event']['end_s']
assert controls[('A2',3,8)]['summary']['requests']['r']['first_token_s']==0

 'external_commits_during_audit':late['commits_during_audit'],'external_source_changes':external_source_changes,
