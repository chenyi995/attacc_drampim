def prepare(policy):
    tlb = wr.CacheBlendTLB(256, 'slice')
    for kind, owner, count in [('master', 'producer', 256), ('diff', 'consumer', 8)]:
        for row in range(count):
            tlb.reserve(0, owner, 'chunk', row, kind)
    tlb.finalize()
    reads = [tlb.locate(0, owner, 'chunk', row, kind)
             for kind, owner, count in [('master', 'producer', 256), ('diff', 'consumer', 8)]
             for row in range(count)]
    return tlb, reads

def run_case(label, policy, kv_heads, heads_per_hbm, q, gqa, command, old=False):
    captured = []
    def output_runs(pim_type, op, power_constraint):
        groups = getattr(op, 'pim_kv_extent_groups', ())
        lengths = ([sum(rows for _, _, rows in extents) for _, _, extents in groups]
                   if groups else [run[2] for run in op.pim_kv_runs])
        captured.append(dict(numOp=op.numOp, m=op.m, n=op.n,
                             shared_queries=op.pim_shared_queries,
                             batch_command=op.pim_batch_command,
                             lane_rows=lengths))
        # Counter-derived traffic is a synthetic linear device fixture.
        return [(rows * 1e-9, [rows * v for v in (32, 16, 8, 4, 64)])
                for rows in lengths]
    acc = PIM.__new__(PIM)
    acc.ramulator = SimpleNamespace(output_runs=output_runs)
    acc.pim_type, acc.power_constraint, acc.num_attacc = PIMType.BA, True, 2
    acc.energy_table = ENERGY_TABLE['PIM'][PIMType.BA]
    acc.io_energy_table = acc.energy_table['io']
    system = SimpleNamespace(devices={'Acc': acc})
    op = Layer('gen', 'score', LayerType.MATMUL, False, DataType.W16A16,
               q, 264, 128, kv_heads)
    op.pim_shared_kv = q * gqa > 1
    op.pim_shared_queries = q * gqa
    op.pim_batch_command = command
    tlb, reads = prepare(policy)
    events = []
    saved = wr._stack_energy_scale
    if old:
        wr._stack_energy_scale = lambda heads, density: -(-heads // density)
    try:
        wr._append_placement_pim_scan(
            system, events, op=op, layer=0, tier=0, request=label, name='pim_kv_scan',
            reads=reads, deps=(), positions=tuple(range(q)), policy=policy,
            heads_per_hbm=heads_per_hbm, tlb=tlb)
    finally:
        wr._stack_energy_scale = saved
    rec = records[-1]
    result = dict(label=label, policy=policy, old_formula=old, kv_heads=kv_heads,
                  heads_per_hbm=heads_per_hbm, q=q, gqa=gqa,
                  device_input=captured[0],
                  event_energy_nj=sum(e.energy_nj for e in events),
                  probe_energy_nj=rec['scan_energy_nj'],
                  event_max_time_s=max(e.time_s for e in events),
                  probe_max_time_s=rec['scan_time_s'],
                  lane_count=len(events))
    result['diagnostic_to_event_energy_ratio'] = result['probe_energy_nj'] / result['event_energy_nj']
    result['folded_rows_over_logical_rows'] = sum(captured[0]['lane_rows']) / len(reads)
    return result

cases = []
configs = [('A1_decode_path', 'slice', 1, 1, 'replicate'),
           ('A3b_decode_path', 'slice-append', 1, 1, 'replicate'),
           ('A4c_decode_path', 'master-diff-local-append', 1, 1, 'replicate'),
           ('A4e_decode_path', 'master-diff-table-local-append', 1, 1, 'replicate'),
           ('A5_prefill_path', 'master-diff-table-local-append', 8, 1, 'mq'),
           ('A6_pim_prefill_path', 'master-diff-table-local-append', 8, 1, 'mq'),
           ('GQA_decode_path', 'master-diff-table-local-append', 1, 4, 'mq'),
           ('GQA_tail_prefill_path', 'master-diff-table-local-append', 2, 4, 'mq')]
for label, policy, q, gqa, command in configs:
    old = run_case(label, policy, 26, 6, q, gqa, command, old=True)
    new = run_case(label, policy, 26, 6, q, gqa, command)
    assert old['device_input'] == new['device_input']
    assert old['event_max_time_s'] == new['event_max_time_s']
    assert abs(new['event_energy_nj'] / old['event_energy_nj'] - 26/30) < 1e-12
    assert new['device_input']['numOp'] == 1
    assert new['device_input']['shared_queries'] == q*gqa
    assert abs(new['diagnostic_to_event_energy_ratio'] - 1.0) < 1e-12
    cases.append(dict(before=old, after=new,
                      energy_after_over_before=new['event_energy_nj']/old['event_energy_nj']))

controls = []
for heads, density in [(8, 8), (8, 4), (26, 1), (30, 6)]:
    before = run_case('divisible_control', 'slice-append', heads, density, 1, 1, 'replicate', old=True)
    after = run_case('divisible_control', 'slice-append', heads, density, 1, 1, 'replicate')
    assert before['event_energy_nj'] == after['event_energy_nj']
    controls.append(after)
