from pathlib import Path

import sys,json,os
sys.dont_write_bytecode=True
ROOT=Path('/data2/chenyi9/KV-PIM/attacc_drampim_822')
sys.path.insert(0,str(ROOT))
from src import workload_runner as wr
from src.devices import PIM
from src.model import Transformer
from src.type import DataType,PIMType
from src.ablation import resolve_config

runs=((0,8388608,256,0,1),(1073741824,1082130432,8,1,1))
groups=((0,1,((0,8388608,256),)),(1,1,((1073741824,1082130432,8),)))
shape=([256.,8.]+[0.]*14,[0,1],runs,groups)
class FixedRamulator:
    def output_runs(self,*args): return [(2e-6,[1.,1.,1.,1.,1.]),(7e-6,[1.,1.,1.,1.,1.])]
class FixedGPU:
    def get_time_and_energy(self,op):return (5e-6 if op.name=='score' else 0.0),(0.,)
acc=PIM.__new__(PIM)
acc.ramulator=FixedRamulator();acc.pim_type=PIMType.BA;acc.power_constraint=True
acc.io_energy_table=[0.,0.,0.,0.];acc.energy_table={'mem':1.,'alu':1.};acc.num_attacc=1;acc.num_hbm=1
model=Transformer(dict(name='scan-max-fixed',ndec=1,num_heads=8,hdim=1024,ff_scale=4,dtype=DataType.W16A16,gqa_size=1),1)
model.build(1,1,2,True)
templates={l.name:l for l in model.sum_decoder}
system=SimpleNamespace(model=model,devices={'Acc':acc,'GPU':FixedGPU()})
wr._EC=None;wr._WARM_BYPASS=None
out={'revision':'da220d1','scope':'No Ramulator/device performance measurement. Injected channel shapes and fixed device costs: channel0=2us, channel1=7us, GPU attention=5us. The PIM converter, placement-event helper, scheduler and A6 reducer are real code.','lane_inputs':[{'channel':r[3],'tokens':r[2],'fixed_service_us':t} for r,t in zip(runs,(2,7))],'rungs':{}}
with patch.object(wr,'_placement_channel_runs',return_value=shape),patch.object(wr.layout_probe,'enabled',return_value=False):
    for rung in ('A3b','A4c','A4e','A5','A6'):
        cfg=resolve_config(rung,None,None,None,policy='recompute')
        policy={'naive':'slice-append','master-diff-local':'master-diff-local-append','master-diff-table-local':'master-diff-table-local-append'}[cfg.kv_mapping]
        events=[];op=deepcopy(templates['score']);op.m=1;op.numOp=8
        scans=wr._append_placement_pim_scan(system,events,op=op,layer=0,tier=0,request='r',name='pim_kv_scan_score_softmax_pv',reads=[],deps=(),positions=(0,),policy=policy,heads_per_hbm=8)
        join=wr._cacheblend_event(events,layer=0,tier=0,request='r',name='die_score_assembly',device='DIE',rows=1,time_s=0.,energy=(),deps=scans,positions=(0,))
        scheduled=wr._schedule_cacheblend(events,pipe=True)
        out['rungs'][rung]={'events':[{'id':e.event_id,'device':e.device,'service_us':e.time_s*1e6,'start_us':e.start_s*1e6,'end_us':e.end_s*1e6,'deps':e.depends_on} for e in scheduled],'join_us':next(e.end_s*1e6 for e in scheduled if e.event_id==join),'lane_sum_us':sum(e.time_s*1e6 for e in scheduled if e.device.startswith('PIM:'))}
        assert out['rungs'][rung]['join_us']==7.0
    tlb=SimpleNamespace(layout_policy='master-diff-table-local-append',scan_runs=lambda locs:runs)
    log=Path('/tmp/scan_max_20260905_choices.jsonl');log.write_text('')
    with patch.dict(os.environ,{'KVPIM_PREFILL_SIDE_LOG':str(log)}):
        side=wr._resolve_prefill_side(system,tlb,templates['comm_x2g'],templates,pim_prefill_mode='dynamic',request_id='fixed',decided={},readback_rows=[],compute_positions=[0],scan_locations=[],heads=8,local_hidden=1024,dbyte=2,batch_size=8,gemv_buffer_bytes=512,pim_batch_command='mq',pim_pe_freq_ghz=1.3)
    out['a6']=json.loads(log.read_text())
    assert side=='gpu' and out['a6']['t_bank_s']==7e-6 and out['a6']['t_xpu_s']==5e-6
Path('/tmp/scan_max_20260905_evidence.json').write_text(json.dumps(out,indent=2)+'\n')
print(json.dumps({'scope':out['scope'],'rungs':{r:{'join_us':v['join_us'],'lane_sum_us':v['lane_sum_us']} for r,v in out['rungs'].items()},'a6':out['a6']},indent=2))
