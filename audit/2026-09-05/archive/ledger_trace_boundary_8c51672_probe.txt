def alloc(reservations):
    t=CacheBlendTLB(256,'slice')
    for owner,fp,rows,kind in reservations:
        for row in rows:t.reserve(0,owner,fp,row,kind)
    t.finalize()
    return t

def reads(t,reservations):
    return [t.locate(0,o,f,r,k) for o,f,rs,k in reservations for r in rs]

def groups(t,rs,p='slice-append',heads=1):
    return t.physical_ledger(p,heads).extent_groups(reads(t,rs))

def decode(addr):
    bits=[5,14,2,2,1,1,4]
    names=['column','row','bank','bankgroup','rank','pseudochannel','channel']
    a=addr>>5
    fields={}
    for name,b in zip(names,bits):
        fields[name]=a&((1<<b)-1)
        a>>=b
    return fields

def trace(extents):
    spec=importlib.util.spec_from_file_location('trace_probe',ROOT/'pim_ramulator_src/trace_gen/gen_trace_attacc_bank.py')
    g=importlib.util.module_from_spec(spec);spec.loader.exec_module(g)
    g.n_channel=1; g.pool_base=extents[0][0]//(1<<30)
    g.Attention(sum(e[2] for e in extents),extents[0][0],extents[0][1],0,valid_channel=1,extents=extents)
    def parse(v):return [[int(cmd.split()[1],16) for cmd in bucket] for bucket in v[0]]
    return {'score':parse(g.cmd_score_mac),'context':parse(g.cmd_context_mac)}
