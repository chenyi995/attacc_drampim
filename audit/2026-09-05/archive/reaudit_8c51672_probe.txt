class Capture(old.Device):
    def __init__(self):
        super().__init__()
        self.ops = []
    def record(self, op):
        self.ops.append({'name':op.name,'m':op.m,'n':op.n,'k':op.k,'numOp':op.numOp,
            'queries':getattr(op,'pim_shared_queries',1),
            'command':getattr(op,'pim_batch_command',None),
            'runs':deepcopy(getattr(op,'pim_kv_runs',())),
            'groups':deepcopy(getattr(op,'pim_kv_extent_groups',()))})
    def get_time_and_energy(self, op):
        self.record(op)
        return super().get_time_and_energy(op)
    def get_time_and_energy_runs(self, op):
        self.record(op)
        return super().get_time_and_energy_runs(op)

def run(wl, rung, gqa=1, batch=8):
    policy='no-reuse' if rung=='A1' else 'recompute'
    plan=build_reuse_plan(wl,policy,epic_prefix_recompute_tokens=1,recompute_canonical=False)
    cfg=resolve_config(rung,None,None,None,policy=policy)
    system=old.system(gqa=gqa)
    system.devices={'GPU':Capture(),'Acc':Capture()}
    rep=w.run_reuse_prefill(system,wl,plan,pipe=True,cacheblend_batch_size=batch,
        include_events=True,warm=False,decode_attn=cfg.decode_attn,
        kv_mapping=cfg.kv_mapping,channel_placement=cfg.channel_placement,
        pim_prefill_mode=cfg.prefill_attn,pim_batch_command=cfg.pim_batch_command,
        pim_pe_freq_ghz=cfg.pim_pe_freq_ghz,gemv_buffer_bytes=cfg.gemv_buffer_bytes)
    return rep,system,plan

def req(rid,segments,lout=2):
    return Request(rid,0,None,lout,tuple(segments),sum(s.length for s in segments))
