def inspect(label,kind,heads,items):
    cls=w.NaiveKVLayout if kind=='A3b' else w.LocalDiffKVLayout
    tlb=cls(256,'slice')
    for fp,t,n in items:
        for r in range(n):tlb.reserve(0,'agent',fp,r,t)
    tlb.finalize()
    ledger=w.PhysicalLedger.build(tlb,tlb.layout_policy,heads)
    answer={'rung':kind,'heads_per_hbm':heads,'stripe':ledger.stripe,'reservation_order':items,'objects':[]}
    for fp,t,n in items:
        key=(0,'agent',fp,t)
        objectkey=ledger.index[key][0]
        rows,indices,slot,start=ledger.objects[objectkey]
        channel=ledger._channel(0,slot)
        answer['objects'].append({'fingerprint':fp,'kind':t,'tokens':n,'head0_channel':channel,'start_byte_within_channel':start,'byte_in_32B_column':start%32,'byte_in_1024B_row':start%1024,'logical_payload_bytes':n*4,'occupied_32B_column_indices':[start//32,(start+n*4-1)//32],'occupied_1024B_row_indices':[start//1024,(start+n*4-1)//1024]})
    return answer
