sources={}
def read(path):
    raw=path.read_bytes()
    sources[str(path)]={'sha256':hashlib.sha256(raw).hexdigest(), 'bytes':len(raw),
        'mtime_utc':datetime.datetime.fromtimestamp(path.stat().st_mtime,datetime.timezone.utc).isoformat()}
    return raw.decode()

    recs=d['summary']['requests']
    spans=[(r,rec['end_s']-rec['first_token_s'],agents[r]['lout']-1) for r,rec in recs.items()]
    total_steps=sum(v[2] for v in spans)
    end=max(rec['end_s'] for rec in recs.values())
    first=min(rec['first_token_s'] for rec in recs.values())
    start=min(b['attention_start_s'] for b in d['batches'] if b.get('attention_start_s') is not None)
    lane_sum=d['pim_pool_time_s_unoverlapped']
    gpu=d['gpu_time_s_unoverlapped']
    rows[rung]={
       'makespan_s':d['makespan_s'],
       'tbt_weighted_us':sum(v[1] for v in spans)/total_steps*1e6,
       'tbt_mean_us':sum(v[1]/v[2] for v in spans)/len(spans)*1e6,
       'pim_lane_sum_s':lane_sum,'gpu_time_sum_s':gpu,
       'pim_energy_j':d['energy_breakdown_nj']['by_class']['PIM']*1e-9,
       'total_energy_j':d['energy_nj']*1e-9,
       'first_decode_attention_s':start,'first_token_completed_s':first,
       'decode_attention_to_end_span_s':end-start,
       'first_token_to_end_span_s':end-first,
       'lane_count_assumption':16,
       'average_lane_busy_over_full_e2e':lane_sum/(16*d['makespan_s']),
       'average_lane_busy_over_decode_attention_window':lane_sum/(16*(end-start)),
       'lane_sum_div_16_div_firsttoken_to_end_proxy':lane_sum/(16*(end-first)),
       'lane_sum_div_e2e_is_not_utilization':lane_sum/d['makespan_s'],
       'gpu_busy_over_full_e2e':gpu/d['makespan_s'],
       'all_pim_removed_e2e_speedup_ceiling_from_unchanged_gpu_work':d['makespan_s']/gpu,
       'all_pim_removed_e2e_reduction_ceiling_from_unchanged_gpu_work':1-gpu/d['makespan_s'],
       'events_retained':d.get('events') is not None,
       'corrected_rows_sha':d['corrected_rows_sha'], 'link_bytes':d['link_bytes'],
       'policy':d['policy'],'batch_size':d['cacheblend_batch_size'],
       'pim_prefill_mode':d['pim_prefill_mode'],'pim_batch_command':d['pim_batch_command'],
       'pim_pe_freq_ghz':d['pim_pe_freq_ghz'],'gemv_buffer_bytes':d['gemv_buffer_bytes'],
       'history_rows':d['history_rows'],'prefill_attention_rows':d['prefill_attention_rows'],
       'ablation':d['ablation'],'ramulator_signature_cache':d['ramulator_signature_cache'],
       'pipeline':d['overlap_validation']['pipe'],
       'model_label_csv':csv_rows[rung]['model'],
       'platform_log_match':re.findall(r'dgx-attacc:.*',log),
       'workload_metadata_sha256':hashlib.sha256(json.dumps(d['workload'],sort_keys=True).encode()).hexdigest()
    }

    assert all(agents[r]['lout']==256 for r in recs)
