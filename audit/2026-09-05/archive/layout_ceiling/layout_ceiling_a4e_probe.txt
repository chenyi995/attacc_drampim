def optimal_two_slots(reads):
    # All masters may be assigned individually per scan; current diffs stay
    # on slot1. This relaxes the one persistent table shared across scans.
    unique={(l.layer,l.owner,l.fingerprint,l.kind,l.owner_row):l for l in reads}
    fixed=sum(l.kind=='diff' for l in unique.values())
    weights=Counter((l.layer,l.owner,l.fingerprint,l.kind,l.owner_row//256)
                    for l in unique.values() if l.kind!='diff')
    total=sum(weights.values())+fixed
    bits=1
    for w in weights.values():bits|=bits<<w
    # slot0 receives x master tokens; slot1 receives all remaining plus diffs.
    target=total//2
    below=(bits&((1<<(target+1))-1)).bit_length()-1
    higher=bits>>(target+1)
    choices=[below]
    if higher:choices.append(target+1+((higher&-higher).bit_length()-1))
    best=min(max(x,total-x) for x in choices)
    return {'relaxed_balanced_lb':(total+1)//2,'per_scan_block_optimum':best,
            'master_tokens':sum(weights.values()),'fixed_diff_tokens':fixed,
            'master_blocks':len(weights)}

def analyse(path):
    wl=load_workload(path);plan=build_reuse_plan(wl,'recompute',epic_prefix_recompute_tokens=8)
    tlbs={'A3b':NaiveKVLayout(256,'slice'),'A4c':LocalDiffKVLayout(256,'slice'),
          'A4e':TableLocalDiffKVLayout(256,'slice')}
    for tlb in tlbs.values():_prepare_cacheblend_tlb(wl,plan,1,tlb,_parent_output_fingerprints(wl))
    entries=[]
    for req in wl.requests:
        binding={r:_cacheblend_tlb_rows(wl,plan,0,req,t) for r,t in tlbs.items()}
        entry={'request':req.request_id,'tier':req.tier,'views':{}}
        for view in ('reused_prefill_ledger_view','full_prompt_decode_view'):
            row={}
            for rung,tlb in tlbs.items():
                locations=[l for _,reused,_,l in binding[rung]
                           if view=='full_prompt_decode_view' or reused]
                reads,_,_=_pool_reads(tlb,locations)
                groups=_striped_append_channel_extents(reads,policy=tlb.layout_policy,heads_per_hbm=8,tlb=tlb)
                loads={ch:sum(c for k,v,c in extents) for ch,n,extents in groups}
                row[rung]={'loads_head0':[loads.get(0,0),loads.get(1,0)],
                           'max_lane_tokens':max(loads.values(),default=0)}
                assert row[rung]['max_lane_tokens']==max(row[rung]['loads_head0'])
                if rung=='A4e':row['lower_bounds']=optimal_two_slots(reads)
            assert sum(row['A4c']['loads_head0'])==sum(row['A4e']['loads_head0'])
            assert row['A4e']['max_lane_tokens']>=row['lower_bounds']['per_scan_block_optimum']
            entry['views'][view]=row
        entries.append(entry)
    summaries={}
    for view in ('reused_prefill_ledger_view','full_prompt_decode_view'):
        def reduce(xs):
            sums={r:sum(e['views'][view][r]['max_lane_tokens'] for e in xs) for r in tlbs}
            lb=sum(e['views'][view]['lower_bounds']['per_scan_block_optimum'] for e in xs)
            denom=sums['A4c']
            return {'requests':len(xs),'sum_max_lane_tokens':sums,'sum_per_scan_block_optimum':lb,
               'a3b_to_a4e_lane_reduction':1-sums['A4e']/sums['A3b'] if sums['A3b'] else None,
               'a4c_to_a4e_lane_reduction':1-sums['A4e']/denom if denom else None,
               'a4c_to_optimistic_lane_reduction':1-lb/denom if denom else None,
               'fraction_of_optimistic_lane_savings_captured':(denom-sums['A4e'])/(denom-lb) if denom!=lb else None,
               'a4c_to_a4e_row_work_speedup':denom/sums['A4e'] if sums['A4e'] else None,
               'a4c_to_optimistic_row_work_speedup':denom/lb if lb else None,
               'a4e_vs_a4c_improved_requests':sum(e['views'][view]['A4e']['max_lane_tokens']<e['views'][view]['A4c']['max_lane_tokens'] for e in xs),
               'a4e_vs_a4c_worse_requests':sum(e['views'][view]['A4e']['max_lane_tokens']>e['views'][view]['A4c']['max_lane_tokens'] for e in xs)}
        worker=[e for e in entries if e['request']!='a0_owner']
        summaries[view]={'all_requests':reduce(entries),'workers':reduce(worker),
                        'last_worker_tier':reduce([e for e in worker if e['tier']==max(v['tier'] for v in worker)])}
    return {'path':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
            'summaries':summaries,'requests':entries}

names=['B1_turns.json','T9_lout_mix_32_turns.json','T9_lout_mix_8_turns.json',

Each request weighted once, no live current-output tokens. Per-scan optimum freely relocates whole master blocks for that one readset, diffs pinned at slot1; optimistic bound, not a globally realizable table.','cases':{}}
