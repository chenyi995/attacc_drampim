def req(rid, segments, tier=0, parent=None, history=0):
    return Request(rid, tier, parent, 2, tuple(segments), sum(s.length for s in segments), history)

def run(wl, rung, policy='epic'):
    if rung == 'A1':
        policy = 'no-reuse'
    p = build_reuse_plan(wl, policy, epic_prefix_recompute_tokens=8,
                         recompute_canonical=PRESETS[rung]['kv_mapping'] != 'naive')
    c = resolve_config(rung, None, None, None, policy=policy)
    s = old.system()
    r = wr.run_reuse_prefill(s, wl, p, pipe=True, cacheblend_batch_size=8,
                            include_events=True, warm=False, decode_attn=c.decode_attn,
                            kv_mapping=c.kv_mapping, channel_placement=c.channel_placement,
                            pim_prefill_mode=c.prefill_attn, pim_batch_command=c.pim_batch_command,
                            pim_pe_freq_ghz=c.pim_pe_freq_ghz, gemv_buffer_bytes=c.gemv_buffer_bytes)
    return r, s, p

def digest(obj):
    return hashlib.sha256(json.dumps(obj, sort_keys=True).encode()).hexdigest()

        'metadata_cost': sum(e['time_s'] + e['energy_nj'] for e in es if e['device'] in ('DIE','TLB')),
    }
