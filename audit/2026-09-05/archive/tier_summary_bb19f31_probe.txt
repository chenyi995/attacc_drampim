import contextlib
import csv
import hashlib
import io
import json
from pathlib import Path
import subprocess
import sys
from types import SimpleNamespace

ROOT = Path('/data2/chenyi9/KV-PIM/attacc_drampim_822')
sys.path.insert(0, str(ROOT))
from src.model import Transformer
from src.type import DataType, DeviceType
from src.workload import Request, Segment, Workload, build_reuse_plan
from src.workload_runner import run_reuse_prefill

DEST = Path('/tmp/tier_summary_bb19f31')
DEST.mkdir(exist_ok=True)
class Device:
    num_hbm = 1
    peak_memory_bandwidth = 1e12
    softmax_peak_bandwidth = 1e12
    energy_table = {'mem':1, 'sram':1}
    def get_time_and_energy(self, layer):
        return .001, [1, 0, 0, 0, 0, 0]
    def get_time_and_energy_runs(self, layer):
        return [(.002,[2,0,0,0,0,0]) for _ in layer.pim_kv_runs]

def system():
    return SimpleNamespace(hetero_name=DeviceType.PIM,
        devices={'GPU':Device(), 'Acc':Device()},
        model=Transformer({'name':'toy','ndec':1,'num_heads':4,'hdim':512,
                           'ff_scale':4,'dtype':DataType.W16A16}, 1))

real_summaries = []
for rung, history, batch in [('A2',0,1), ('A3b',0,1), ('A2',3,1), ('A3b',3,1),
                              ('A2',0,8), ('A3b',0,8), ('A2',3,8), ('A3b',3,8)]:
    request = Request('r',0,None,1,(Segment('sys','s',4),),4,history_len=history)
    workload = Workload('rag',(request,),{})
    plan = build_reuse_plan(workload,'recompute',epic_prefix_recompute_tokens=1)
    with contextlib.redirect_stdout(io.StringIO()):
        report = run_reuse_prefill(system(),workload,plan,pipe=True,
            decode_attn='gpu' if rung=='A2' else 'pim',pim_prefill_mode='gpu',
            kv_mapping='none' if rung=='A2' else 'naive',channel_placement='slice',include_events=True,
            cacheblend_batch_size=batch)
    events = report['events']
    row = dict(rung=rung,history_len=history,cacheblend_batch_size=batch,summary=report['summary'],
               final_gpu_event=max((e for e in events if e['device']=='GPU'), key=lambda e:e['end_s']),
               final_non_decode_events=[e for e in events if not e['name'].startswith('decode_') and
                                         e['query_positions']==[request.total_length]],
               decode_events=[dict(name=e['name'],start_s=e['start_s'],end_s=e['end_s'],
                                   query_positions=e.get('query_positions'),
                                   batch_members=e.get('batch_members'))
                              for e in events if e['name'].startswith('decode_')])
    real_summaries.append(row)
    (DEST / f'real_{rung}_history{history}_batch{batch}.json').write_text(json.dumps(report,indent=2)+'\n')

    'c':{'tier':1,'prefill_end_s':14.,'first_token_s':15.,'end_s':20.}}}
for rung in ('A1','A2','A3b','A4c','A4e','A5','A6'):
    batches = ([] if rung=='A2' else [

    report={'summary':summary,'batches':batches,'makespan_s':20.,'energy_nj':1.,
            'energy_breakdown_nj':{'by_class':{}}}
