def reserve(tlb, owner, fp, count, kind='master'):
    for i in range(count):
        tlb.reserve(0, owner, fp, i, kind)

def locations(tlb, owner, fp, count, kind='master'):
    return [tlb.locate(0, owner, fp, i, kind) for i in range(count)]

def store_events(locs):

        request='audit', name='store', locations=locs, dbyte=2, deps=(), positions=())
    return [{'device': e.device, 'rows': e.rows, 'time_s': e.time_s,
             'first_key_address': e.dram_addresses[0]} for e in es]

# Exercise the existing A6 chooser with a deliberately inspectable device
# pricing function. It proves which lane is queried, not real hardware cost.
class PriceGPU(old.Device):
    def get_time_and_energy(self, op):
        self.calls.append((op.name, op.m, op.n, op.k, op.numOp))
        return (50e-6 if op.name in ('score', 'softmax', 'context') else 20e-6), [0]

class PricePIM(old.Device):
    def get_time_and_energy_runs(self, op):
        groups = getattr(op, 'pim_kv_extent_groups', ())
        self.calls.append({'m': op.m, 'n': op.n, 'numOp': op.numOp,
                           'groups': deepcopy(groups)})
        return [(len(extents) * 100e-6 + sum(r[2] for r in extents)*1e-9, [0])
                for ch, count, extents in groups]
